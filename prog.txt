# 1. Declarative Rewrite Rules
rule my_double_angle: 2 * sin(?t) * cos(?t) <=> sin(2 * ?t);
rule my_trig_identity: cos(?u)^2 + sin(?u)^2 => 1;
rule distrib: ?a * (?b + ?c) <=> ?a * ?b + ?a * ?c;

# 2. Automated Formal Equivalence Proofs (E-Graph Congruence Closure)
